    instance method Pathname#dirname

    dirname -> Pathname[permalink][rdoc][edit]
    () → Pathname

    Pathname.new(File.dirname(self.to_s)) と同じです。

    例 
    require "pathname"

p Pathname('/usr/bin/shutdown').dirname # => #<Pathname:/usr/bin>
