    instance method Net::IMAP::Config#inspect

    inspect -> StringRuby 4.0 から[permalink][rdoc][edit]
    to_s -> String

    self で値を設定した属性と継承の連鎖を表す文字列を返します。

    継承元で値を設定した属性も、継承元ごとに再帰的に示します。既定値の設定から継承した属性は示しません。

    例 
    require 'net/imap'

Net::IMAP.debug = true
p Net::IMAP::Config.global
# => #<Net::IMAP::Config.global debug=true inherits from Net::IMAP::Config.default>
